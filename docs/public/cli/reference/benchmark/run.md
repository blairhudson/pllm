  -h, --help            show this help message and exit
  --format {human,json,jsonl}
                        result format (default: human)
  --quiet               suppress non-error diagnostics
  --no-color            disable colored output
  --no-input            fail instead of prompting
  --dry-run             report work without persistent writes or service

                        targets

  --model-id MODEL_ID   stable model identity stored in the report
  --tiny                use generated random weights for a transport smoke
                        test

  --prompt-file PROMPT_FILE
                        read prompt text from this file
  --max-output-tokens MAX_OUTPUT_TOKENS
                        maximum generated tokens per run (default: 24)
  --warmups WARMUPS     warmup runs retained in the report (default: 0)
  --repetitions REPETITIONS
                        measured runs (default: 1)
  --timeout TIMEOUT     startup and per-run timeout in seconds (default: 900)
  --show-dashboard      open the local dashboard in a browser (default:
                        hidden)
  --output OUTPUT       write the sanitized JSON report
  --save-best PATH      write the Experiment with lowest measured median full
                        latency as canonical JSON
  --force               replace --output if it exists
